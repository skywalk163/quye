### Task 1：语言包 zip 打包，Worker 一次拉取解包

**Files:**
- Modify: `G:\comatespec\quye\建站.py`（`抽取语言包()` 生成清单后追加打 zip）
- Modify: `G:\comatespec\quye\静态\执行器.js`（`准备语言()` 改为拉 zip 解包，逐文件 fetch 作兜底）

**背景**：M2 已知问题——`准备语言()` 按 `清单.json` 逐个 fetch，极快 394 个文件 → 几百次串行请求。改成构建期打成单个 zip，Worker 用 Pyodide 的 `unpack_archive` 一次解包。

- [ ] **Step 1：建站.py 生成清单后打 zip**

在 `建站.py` 的 `抽取语言包()` 里，生成 `清单.json` 的 for 循环之后追加（缩进与函数体一致）：

```python
    # 每门语言额外打一个 zip，供 Worker 一次拉取解包（M2 遗留：394 文件串行 fetch 过慢）
    import zipfile
    for 名 in ["极快", "段言", "光明", "知行"]:
        目录 = py目标 / 名
        if not 目录.exists():
            continue
        zip路径 = py目标 / f"{名}.zip"
        with zipfile.ZipFile(zip路径, "w", zipfile.ZIP_DEFLATED) as zf:
            for p in sorted(目录.rglob("*")):
                if p.is_file() and p.name != "清单.json":
                    zf.write(p, p.relative_to(目录).as_posix())
        print(f"    {名}.zip 打包 {zip路径.stat().st_size // 1024} KB")
```

说明：zip 内路径相对语言目录（如 `jikuai/main.py`、`stdlib/blocks/索引.json`），与 `清单.json` 的相对路径一致；不含 `清单.json` 本身。

- [ ] **Step 2：执行器.js 改用 zip 解包**

替换 `静态/执行器.js` 的 `准备语言()` 函数体中「从 `/静态/py/<语言>/` 抓取 py 文件并挂到 Pyodide FS」那段（当前逐文件 fetch 的 for 循环），改为：

```javascript
  // 从 /静态/py/<语言>.zip 一次拉取 → Pyodide unpack_archive 解包到 /<语言>
  // （M2 遗留：逐文件 fetch 极快 394 个太慢）；zip 拉取失败时回退逐文件清单
  const zip地址 = `/静态/py/${encodeURIComponent(语言)}.zip`;
  const 目标根 = `/${语言}`;
  py.FS.mkdirTree(目标根);
  let 用zip = false;
  try {
    const resp = await fetch(zip地址);
    if (resp.ok) {
      const buf = new Uint8Array(await resp.arrayBuffer());
      py.unpackArchive(buf, "zip", { extractDir: 目标根 });
      用zip = true;
    }
  } catch (e) {
    postMessage({ 类型: "进度", 消息: `zip 解包失败，回退逐文件：${e.message}` });
  }
  if (!用zip) {
    const 根 = `/静态/py/${encodeURIComponent(语言)}/`;
    const 清单 = await (await fetch(`${根}清单.json`)).json();
    for (const 相对路径 of 清单.文件) {
      const resp = await fetch(`${根}${相对路径}`);
      if (!resp.ok) throw new Error(`加载失败：${根}${相对路径}`);
      const py路径 = `${目标根}/${相对路径}`;
      py.FS.mkdirTree(py路径.split('/').slice(0, -1).join('/'));
      py.FS.writeFile(py路径, await resp.text(), { encoding: "utf8" });
    }
  }
```

注意：`清单.json` 仍保留（回退路径与调试用），只是不再是主路径。`sys.path.insert` 与极快热补丁那两段保持不动（Task 2 会替换热补丁）。

```python
"""抽取学习素材：段言 tutorial exercises + 光明 L1_baihua。

产物落到 出/数据/学习素材/<语言>/<关卡>/：
- 题干.md      （段言有；光明用源码首部注释，无独立题干时留空）
- 参考答案.txt  （段言取 <关卡>_solution.duan；光明取 <关卡>.light 源码本身）
- 骨架.txt      （段言有 <关卡>.duan 起始骨架；光明无，留空）
关卡元数据来自 数据/学习清单.json，由建站.py 读入后传进来。
"""
import json
import shutil
from pathlib import Path


def 抽取(段言根: Path, 光明根: Path, 素材目标: Path, 清单: dict) -> None:
    素材目标.mkdir(parents=True, exist_ok=True)

    # 段言：exercises 三件套
    段言练习 = 段言根 / "tools" / "tutorial" / "exercises"
    for 条 in 清单["段言"]:
        关卡 = 条["关卡"]
        目标 = 素材目标 / "段言" / 关卡
        目标.mkdir(parents=True, exist_ok=True)
        题干 = 段言练习 / f"{关卡}.md"
        答案 = 段言练习 / f"{关卡}_solution.duan"
        骨架 = 段言练习 / f"{关卡}.duan"
        if not 答案.exists():
            raise FileNotFoundError(f"段言参考答案缺失：{答案}")
        _写(目标 / "题干.md", 题干)
        _写(目标 / "参考答案.txt", 答案)
        _写(目标 / "骨架.txt", 骨架)

    # 光明：L1_baihua/<关卡>.light（源码即参考答案；题干取源码首部注释）
    光明课 = 光明根 / "examples" / "L1_baihua"
    for 条 in 清单["光明"]:
        关卡 = 条["关卡"]
        目标 = 素材目标 / "光明" / 关卡
        目标.mkdir(parents=True, exist_ok=True)
        源 = 光明课 / f"{关卡}.light"
        if not 源.exists():
            raise FileNotFoundError(f"光明课程缺失：{源}")
        _写(目标 / "参考答案.txt", 源)
        # 题干：取源码开头连续的 # 注释行
        题干行 = []
        for line in 源.read_text(encoding="utf-8").splitlines():
            if line.startswith("#"):
                题干行.append(line.lstrip("# ").rstrip())
            elif line.strip() == "":
                continue
            else:
                break
        (目标 / "题干.md").write_text("\n\n".join(题干行), encoding="utf-8")


def _写(目标: Path, 源: Path) -> None:
    if 源.exists():
        shutil.copy2(源, 目标)
    else:
        目标.write_text("", encoding="utf-8")
```

```python
def 预跑学习基线(数据目录: Path, py根: Path) -> None:
    """跑每关参考答案，抓 stdout 作判对基线；任一关跑不出结果则构建失败。

    部分关卡（如段言 files 的真实文件 IO）在子进程可跑但浏览器 Pyodide 未必支持，
    这类关卡标 在线可判=False，页面只读展示、不提供在线判对。
    """
    # ROOT 是 预跑器.py 所在的仓库根（文件顶部已定义），学习清单是源数据不在产物里
    清单 = json.loads((ROOT / "数据" / "学习清单.json").read_text(encoding="utf-8"))
    素材根 = 数据目录 / "学习素材"
    基线 = {"段言": [], "光明": []}
    失败 = []
    for 语言 in ("段言", "光明"):
        for 条 in 清单[语言]:
            关卡 = 条["关卡"]
            答案文件 = 素材根 / 语言 / 关卡 / "参考答案.txt"
            源码 = 答案文件.read_text(encoding="utf-8")
            r = 跑一个(语言, 源码, py根 / 语言)
            if "error" in r:
                失败.append(f"学习 {语言} · {关卡}: {r['error'][:200]}")
                # 跑不通的关卡：不阻断，标记不可在线判（页面只读展示）
                基线[语言].append({**条, "预期stdout": "", "在线可判": False})
                continue
            基线[语言].append({**条, "预期stdout": r["stdout"], "在线可判": True})
    可判数 = sum(1 for 语言 in 基线 for x in 基线[语言] if x["在线可判"])
    if 可判数 < 15:  # spec § 4.3 要 19 关；留 4 关容差给浏览器不支持的 IO 类
        for 条 in 失败:
            print(f"  ! 学习基线失败: {条}")
        raise RuntimeError(f"学习基线可判关卡仅 {可判数}，低于阈值 15")
    for 条 in 失败:
        print(f"  ! 学习关卡不可在线判（只读展示）: {条}")
    (数据目录 / "学习基线.json").write_text(
        json.dumps(基线, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(f"  学习基线：可在线判 {可判数} / 19 关")
```

说明：阈值 15 是务实兜底——若段言 `files` 等 IO 类关卡在子进程也依赖真实文件而无法稳定预跑，不至于卡死整条构建；但要求至少 15 关能判，保证学习闭环有效。**实测若 19 关全过，可把阈值提到 19 收紧门禁**（见 Step 3 决策）。

```python
"""按 数据/文档清单.json 抽各仓库精选 markdown 到 出/数据/文档/<slug>.md。

源文件缺失即报错——宁可构建失败，也不产出指向死链的文档索引。

import shutil
from pathlib import Path


def 抽取(仓库根映射: dict, 目标: Path, 清单: dict) -> list:
    """仓库根映射: {语言: 仓库根 Path}。返回成功抽取的条目列表（供构建索引）。"""
    目标.mkdir(parents=True, exist_ok=True)
    成功 = []
    缺失 = []
    for 条 in 清单["文档"]:
        根 = 仓库根映射[条["语言"]]
        源 = 根 / 条["源路径"]
        if not 源.exists():
            缺失.append(f'{条["语言"]}:{条["源路径"]}')
            continue
        shutil.copy2(源, 目标 / f'{条["slug"]}.md')
        成功.append(条)
    if 缺失:
        raise FileNotFoundError("文档源缺失（清单需更新或上游改名）：" + "; ".join(缺失))
    return 成功
```

```python
"""极快选块命中率实测（spec §4.5 风险 B）。

在 出/静态/py/极快 布局上跑 retrieve，人工标注的期望块名命中 top-k 即算命中。
用法：先 python 建站.py 产出语言包，再 python 工具/命中率实测.py
"""
import os
import sys
from pathlib import Path

基 = Path(__file__).resolve().parent.parent / "出" / "静态" / "py" / "极快"
os.environ["JIKUAI_PATH"] = str(基 / "stdlib")
sys.path.insert(0, str(基))
from jikuai.ai.retrieval import retrieve  # noqa: E402

# 30 条真实需求 → 期望命中的块名关键词（子串匹配，命中 top-5 即算过）
用例 = [
    ("月薪两万个税多少", "个税"),
    ("增值税怎么算", "增值税"),
    ("等额本息月供", "月供"),
    # ... 补齐到 30 条，覆盖财务/数据/日期/字符串/农历等领域
]


def 主():
    top = 5
    命中 = 0
    漏 = []
    for 需求, 期望 in 用例:
        try:
            结果 = retrieve(需求, top=top)
        except Exception as e:
            漏.append((需求, f"异常:{e}"))
            continue
        名单 = [h.name for h in 结果]
        if any(期望 in n for n in 名单):
            命中 += 1
        else:
            漏.append((需求, 名单))
    总 = len(用例)
    率 = 命中 / 总 if 总 else 0
    print(f"命中 {命中}/{总} = {率:.0%}")
    for 需求, 名单 in 漏:
        print(f"  漏: {需求} -> {名单}")
    return 率


if __name__ == "__main__":
    率 = 主()
    sys.exit(0 if 率 >= 0.6 else 1)
```
