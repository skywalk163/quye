```python
"""从极快仓库抽取核心链路。

来源：src/jikuai/ + stdlib/ + tools/ai-bridge/glue.py + --check/索引.json
去除：pkg/sources.py（git subprocess）
"""
import shutil
from pathlib import Path


def 抽取(仓库根: Path, 目标: Path) -> None:
    目标.mkdir(parents=True, exist_ok=True)

    # 核心 Python 包
    src = 仓库根 / "src" / "jikuai"
    if not src.exists():
        raise FileNotFoundError(f"极快源码目录不存在：{src}")
    shutil.copytree(src, 目标 / "jikuai", dirs_exist_ok=True)

    # 移除浏览器碰不到的模块
    坑 = 目标 / "jikuai" / "pkg" / "sources.py"
    if 坑.exists():
        # 用空实现替换，避免 evaluator 里的可选 import 报错
        坑.write_text(
            '"""在浏览器环境中被替换为空 stub。原文件走 git subprocess。"""\n'
            'def _unavailable(*a, **kw):\n'
            '    raise RuntimeError("包管理器在浏览器中不可用")\n',
            encoding="utf-8"
        )

    # stdlib
    stdlib源 = 仓库根 / "stdlib"
    if stdlib源.exists():
        shutil.copytree(stdlib源, 目标 / "stdlib", dirs_exist_ok=True)

    # AI 桥的 glue 模块
    glue = 仓库根 / "tools" / "ai-bridge" / "glue.py"
    if glue.exists():
        shutil.copy2(glue, 目标 / "glue.py")

    # 索引文件
    索引 = 仓库根 / "--check" / "索引.json"
    if 索引.exists():
        shutil.copy2(索引, 目标.parent.parent.parent / "数据" / "极快索引.json")
```

```python
来源：src/zhixing/compiler/*、src/zhixing/runtime/*、src/zhixing/stdlib/*
知行运行时改动：给 stdlib/loader.py 打补丁，跳过 c_lib/net_lib/db_lib 加载失败（M2 关键补丁点）
排除：pkg_manager.py（subprocess）

import shutil

def 抽取(仓库根: Path, 目标: Path) -> None:
    目标.mkdir(parents=True, exist_ok=True)
    src = 仓库根 / "src" / "zhixing"

        if p.name in 跳过 or p.name == "pkg_manager.py":

                r"(lambda mod=\1: mod)()",  # no-op wrap 用于占位，下面正则会替换真正的调用

                count=0,
            )

                r"['\"]zhixing\.stdlib\.(c_lib|net_lib|db_lib)['\"],?\s*\n?",
                "",
                文本,
            )
            新文本 = 新文本.rstrip() + "\n\n# quye-patched: 移除浏览器不可用的 stdlib 项\n"
```

```python
BUILD_CACHE = ROOT / ".build_cache"


def 浅克隆(名: str, url: str) -> Path:
    """浅克隆到 .build_cache/<名>/，已存在则 pull 更新。"""
    目标 = BUILD_CACHE / 名
    if 目标.exists():
        # 已存在：拉最新
        r = subprocess.run(
            ["git", "-C", str(目标), "fetch", "--depth", "1", "origin", "HEAD"],
            capture_output=True, text=True, timeout=120
        )
        if r.returncode != 0:
            print(f"  ! 更新 {名} 失败：{r.stderr.strip()[:200]}")
            return 目标
        subprocess.run(
            ["git", "-C", str(目标), "reset", "--hard", "FETCH_HEAD"],
            capture_output=True, text=True, timeout=60
        )
        return 目标
    目标.parent.mkdir(parents=True, exist_ok=True)
    r = subprocess.run(
        ["git", "clone", "--depth", "1", url, str(目标)],
        capture_output=True, text=True, timeout=300
    )
    if r.returncode != 0:
        raise RuntimeError(f"克隆 {名} 失败：{r.stderr.strip()[:200]}")
    return 目标
```

```javascript
// Pyodide 加载器：自托管主 + CDN 兜底并发
// 谁先到用谁。返回 Promise<Pyodide>.
export const PYODIDE_VERSION = "0.26.4";
const 自托管 = "/静态/pyodide/";
const CDN = `https://cdn.jsdelivr.net/pyodide/v${PYODIDE_VERSION}/full/`;

let _缓存Promise = null;

export function 加载Pyodide(进度回调) {
  if (_缓存Promise) return _缓存Promise;
  _缓存Promise = _并发拉取(进度回调);
  return _缓存Promise;
}
```

```javascript
// 消息协议：
//   主线程 → Worker: {类型:"跑", 语言, 源码, 会话id}
//   Worker → 主线程: {类型:"结果"|"错误"|"进度", 会话id, ...}

const 自托管 = "/静态/pyodide/";
const CDN = `https://cdn.jsdelivr.net/pyodide/v${self.PYODIDE_VERSION}/full/`;

async function 初始化Pyodide() {
  if (_py) return _py;
  postMessage({ 类型: "进度", 消息: "加载 Pyodide runtime..." });
  // 直接尝试自托管，失败退 CDN。Worker 里 dynamic import 需要绝对 URL
  const 试 = async (base) => {
    self.importScripts(base + "pyodide.js");
    return await self.loadPyodide({ indexURL: base });
  };
  try {
    _py = await 试(new URL(自托管, self.location.href).href);
  } catch (e1) {
    postMessage({ 类型: "进度", 消息: `自托管失败，回退 CDN: ${e1.message}` });
    _py = await 试(CDN);
  }
  return _py;
}

    const resp = await fetch(url);
    if (!resp.ok) throw new Error(`加载失败：${url}`);
    const 内容 = await resp.text();

    py.FS.writeFile(py路径, 内容, { encoding: "utf8" });

  py.runPython(`

if '/${语言}' not in sys.path:
    sys.path.insert(0, '/${语言}')
`);

}
```

```javascript
let _会话id = 0;
let _会话回调 = new Map();
const 超时毫秒 = 5000;

_worker.addEventListener('message', (e) => {
  const { 会话id, 类型 } = e.data;
  if (类型 === '进度') {

function 跑代码(语言, 源码) {
  return new Promise((resolve, reject) => {
    const id = ++_会话id;

      if (data.类型 === '结果') resolve(data);

        _worker.terminate();

}
```

写入 `预跑器.py`：

```python
"""构建期跑官方示例集，产出示例结果.json。

任一示例挂掉，整个构建失败——这是 spec § 4.4 门禁 #1（示例回归）。
"""
import io
import json
import subprocess
import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parent


def 跑一个(语言: str, 源码: str, 语言目录: Path) -> dict:
    """在子进程里跑语言的解释器，避免污染当前进程。"""
    # 用 Python 直接跑对应桥接代码。语言目录已含 py 包。
    桥 = {
        "极快": """
import sys
sys.path.insert(0, r'{lang_dir}')
import io
from jikuai.main import run_source
_buf = io.StringIO()
_old = sys.stdout
sys.stdout = _buf
try:
    run_source({src!r})
finally:
    sys.stdout = _old
print("__STDOUT__", _buf.getvalue(), sep="")
""",
        "段言": """
import sys
sys.path.insert(0, r'{lang_dir}')
import io
from duan_parser_v3 import DuanParser
from code_generator import PythonCodeGenerator
_buf = io.StringIO()
_old = sys.stdout
sys.stdout = _buf
try:
    _mod = DuanParser().parse({src!r})
    _py = PythonCodeGenerator().generate(_mod)
    exec(_py, {{'__name__': '__main__'}})
finally:
    sys.stdout = _old
print("__STDOUT__", _buf.getvalue(), sep="")
""",
        "光明": """
import sys
sys.path.insert(0, r'{lang_dir}')
import io
from light_parser_v3 import LightParser
from code_generator import PythonCodeGenerator
_buf = io.StringIO()
_old = sys.stdout
sys.stdout = _buf
try:
    _ast = LightParser().parse({src!r})
    _py = PythonCodeGenerator().generate(_ast)
    exec(_py, {{'__name__': '__main__'}})
finally:
    sys.stdout = _old
print("__STDOUT__", _buf.getvalue(), sep="")
""",
        "知行": """
import sys
sys.path.insert(0, r'{lang_dir}')
import io
from zhixing.runtime.evaluator import evaluate
from zhixing.compiler.parser import parse
_buf = io.StringIO()
_old = sys.stdout
sys.stdout = _buf
try:
    evaluate(parse({src!r}))
finally:
    sys.stdout = _old
print("__STDOUT__", _buf.getvalue(), sep="")
""",
    }
    if 语言 not in 桥:
        return {"error": f"未知语言 {语言}"}
    code = 桥[语言].format(lang_dir=str(语言目录), src=源码)
    r = subprocess.run(
        [sys.executable, "-c", code],
        capture_output=True, text=True, timeout=30, encoding="utf-8"
    )
    if r.returncode != 0:
        return {"error": r.stderr[:500]}
    # 提取 __STDOUT__ 后的内容
    marker = "__STDOUT__"
    idx = r.stdout.rfind(marker)
    stdout = r.stdout[idx + len(marker):] if idx >= 0 else ""
    return {"stdout": stdout}


def 预跑(数据目录: Path, py根: Path) -> None:
    集 = json.loads((数据目录 / "示例集.json").read_text(encoding="utf-8"))
    结果 = {"工作台官方示例": []}
    失败 = []
    for 条 in 集["工作台官方示例"]:
        r = 跑一个(条["语言"], 条["源码"], py根 / 条["语言"])
        if "error" in r:
            失败.append(f"{条['语言']} · {条['标题']}: {r['error'][:200]}")
            continue
        结果["工作台官方示例"].append({**条, "stdout": r["stdout"]})
    if 失败:
        for 条 in 失败:
            print(f"  ! 预跑失败: {条}")
        raise RuntimeError(f"预跑器失败 {len(失败)} 个示例")
    (数据目录 / "示例结果.json").write_text(
        json.dumps(结果, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print(f"  预跑通过 {len(结果['工作台官方示例'])} 个示例")


if __name__ == "__main__":
    预跑(Path("出/数据"), Path("出/静态/py"))
```
